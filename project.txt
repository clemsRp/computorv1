
#[derive(Debug, Clone, Copy)]
pub struct Fraction {
	num: i128,
	denum: i128,
}

fn gcd(mut a: i128, mut b: i128) -> i128 {
	a = a.abs();
	b = b.abs();

	while b != 0 {
		let r = a % b;
		a = b;
		b = r;
	}

	a.max(1)
}

impl Fraction {
	pub fn new(num: i128, denum: i128) -> Self {
		let divisor = gcd(num, denum);

        Fraction {
            num: num / divisor,
            denum: denum / divisor,
        }
	}

	pub fn from_decimale(base_digits: String, frac_digits: String) -> Self {
		let power: u32 = frac_digits.len().try_into().unwrap();
		let scale = 10_i128.pow(power);

		let base_int: i128 = base_digits.parse().unwrap();
        let frac_int: i128 = if frac_digits.is_empty() {
            0
        } else {
            frac_digits.parse().unwrap()
        };

        let raw_num = base_int * scale + frac_int;
        let raw_denum = scale;

		Self::new(raw_num, raw_denum)
	}

	pub fn default() -> Self {
		Self::new(0, 1)
	}

	pub fn one() -> Self {
		Self::new(1, 1)
	}

	pub fn get_num(&self) -> i128 {
		self.num
	}

	pub fn get_denum(&self) -> i128 {
		self.denum
	}

	pub fn is_neg(&self) -> bool {
		self.num < 0
	}

	pub fn is_zero(&self) -> bool {
		self.num == 0
	}

	pub fn negative(&self) -> Self {
		Self{num: -self.get_num(), denum: self.get_denum()}
	}

	pub fn reciprocal(&self) -> Self {
		Self{num: self.get_denum(), denum: self.get_num()}
	}

	pub fn add(&self, other: &Fraction) -> Self {
		let new_num = self.num * other.get_denum() + other.get_num() * self.denum;
		let new_denum = self.denum * other.get_denum();

		Self::new(new_num, new_denum)
	}

	pub fn sub(&self, other: &Fraction) -> Self {
		self.add(&other.negative())
	}

	pub fn mul(&self, other: &Fraction) -> Self {
		let new_num = self.num * other.get_num();
		let new_denum = self.denum * other.get_denum();

		Self::new(new_num, new_denum)
	}

	pub fn div(&self, other: &Fraction) -> Self {
		self.mul(&other.reciprocal())
	}

	pub fn pow(&self, power: u32) -> Self {
		let new_num = self.num.pow(power);
		let new_denum = self.denum.pow(power);

		Self::new(new_num, new_denum)
	}

	pub fn get_number(&self) -> String {
		(self.num as f64 / self.denum as f64).to_string()
	}
}

#[derive(Debug)]
pub enum TokenKind {
    Number(Fraction),
    X,
    Plus,
    Minus,
    Star,
    Slash,
    Caret,
    LParen,
    RParen,
    Equal,
    End,
}

pub struct Token {
    kind: TokenKind,
    text: String,
}

impl Token {
	pub fn get_kind(&self) -> &TokenKind {
		&self.kind
	}

	pub fn get_text(&self) -> &String {
		&self.text
	}
}

pub fn tokenize(input: String) -> Vec<Token> {
    let raw_tokens: Vec<char> = input.chars().collect();
    let mut tokens: Vec<Token> = Vec::new();

    let mut i = 0;
    while i < raw_tokens.len() {
        // Get token datas
        let c = raw_tokens[i];
        let (kind, len) = match c {
			// Skip spaces
			' ' | '\t' | '\n' | '\r' => {
			    i += 1;
			    continue;
			}

			// Numbers
			'0'..='9' => {
				let (value, len) = read_number(&raw_tokens, i);
				(TokenKind::Number(value), len)
			}
			
			// Variables
			'X' | 'x' => (TokenKind::X, 1),

			// Operators
			'+' => (TokenKind::Plus, 1),
			'-' => (TokenKind::Minus, 1),
			'*' => (TokenKind::Star, 1),
			'/' => (TokenKind::Slash, 1),
			'^' => (TokenKind::Caret, 1),
			'=' => (TokenKind::Equal, 1),
			'(' => (TokenKind::LParen, 1),
			')' => (TokenKind::RParen, 1),

			// Invalid char
			_ => panic!("Invalid character : '{c}'"),
        };

        // Add new token
        let text = raw_tokens[i..i + len].iter().collect();
        tokens.push(Token {kind: kind, text: text});

        i += len;
    }

	// Terminate tokens
	tokens.push(Token{kind: TokenKind::End, text: String::new()});

    tokens
}

fn read_number(raw_tokens: &[char], start: usize) -> (Fraction, usize) {
	let mut base_digits = String::new();
	let mut frac_digits = String::new();

	let t_len = raw_tokens.len();
	let mut i = start;

	// Get base digits
	while i < t_len && raw_tokens[i].is_ascii_digit() {
		base_digits.push(raw_tokens[i]);
		i += 1;
	}

	if i < t_len && raw_tokens[i] == '.' {
		// Get frac digits
		i += 1;
		while i < t_len && raw_tokens[i].is_ascii_digit() {
			frac_digits.push(raw_tokens[i]);
			i += 1;
		}

		// Check non empty list
		// if frac_digits.is_empty() {
			// TODO
		// }
	}

	// if i < t_len && raw_tokens[i] == '.' {
		// TODO
	// }

	let value = Fraction::from_decimale(base_digits, frac_digits);

	(value, i - start)
}

	tokens: Vec<Token>,
	index: usize,
}

pub struct Equation {
	left: Polynomial,
	right: Polynomial,
}

impl Parser {
	pub fn new(tokens: Vec<Token>) -> Self {
		Parser{tokens: tokens, index: 0}
	}

	fn peek(&self) -> Token {
		self.tokens[self.index]
	}

	fn seek(&self) -> Token {
		let token = self.tokens[self.index].clone();
        if token.kind != TokenKind::End {
            self.index += 1;
        }
        token
	}

	pub fn parse_expr() -> Polynomial {

	}
}

pub fn parse_equation(input: String) -> Equation {
	let tokens = tokenize(input);
	let parser = Parser::new(tokens);
	let left = parser.parse_expr();
	parser.expect_equal()?;
	let right = parser.parse_expr();
	parser.expect_end()?;

	Equation {
		left: left,
		right: right,
	}
}
use fraction::Fraction;

#[derive(Debug, Clone)]
pub struct Polynomial {
	coeffs: Vec<Fraction>,
}

impl Polynomial {
	fn normalized(raw_coeffs: Vec<Fraction>) -> Self {
		// Get last valid coeff
		let mut last_index = raw_coeffs.len();
		while last_index - 1 > 0 && raw_coeffs[last_index-1].is_zero() {
			last_index -= 1;
		}

		// Handle empty coeffs
		if raw_coeffs.len() == 0 {
			return Polynomial{coeffs: raw_coeffs};
		}

		// Get valid coeffs
		let mut i = 0;
		let mut coeffs: Vec<Fraction> = Vec::new();
		while i <= last_index - 1 {
			coeffs.push(raw_coeffs[i].clone());
			i += 1;
		}

		Polynomial{coeffs: coeffs}
	}

	pub fn new(raw_coeffs: Vec<Fraction>) -> Self {
		Self::normalized(raw_coeffs)
	}

	fn get_coeffs(&self) -> &Vec<Fraction> {
		&self.coeffs
	}

	pub fn constant(value: Fraction) -> Self {
		Self::normalized(vec![value])
	}

	pub fn is_constant(&self) -> bool {
		self.get_coeffs().len() <= 1
	}

	pub fn x(coeff: Fraction, power: usize) -> Self {
		let mut raw_coeffs = vec![Fraction::default();power];
		raw_coeffs.push(coeff);
		Self::normalized(raw_coeffs)
	}

	pub fn negative(&self) -> Self {
		// Get fractions
		let coeffs: &Vec<Fraction> = self.get_coeffs();
		
		let mut res: Vec<Fraction> = vec![Fraction::default();coeffs.len()];

		// Get fractions reciprocal
		for i in 0..coeffs.len() {
			res[i] = coeffs[i].negative();
		}

		Self::normalized(res)
	}

	pub fn reciprocal(&self) -> Self {
		// Get fractions
		let coeffs: &Vec<Fraction> = self.get_coeffs();
		
		let mut res: Vec<Fraction> = vec![Fraction::default();coeffs.len()];

		// Get fractions reciprocal
		for i in 0..coeffs.len() {
			res[i] = coeffs[i].reciprocal();
		}

		Self::normalized(res)
	}

	pub fn add(&self, other: &Polynomial) -> Self {
		// Get fractions
		let left = self.get_coeffs();
		let right = other.get_coeffs();
		
		// Get res len
		let l_len = left.len();
		let r_len = right.len();
		let length = l_len.max(r_len);
		
		let mut res: Vec<Fraction> = vec![Fraction::default();length];

		// Fill res
		for i in 0..length {
			let mut temp = Fraction::default();
			if i < l_len {
				temp = temp.add(&left[i]);
			}
			if i < r_len {
				temp = temp.add(&right[i]);
			}

			res[i] = temp;
		}

		Self::normalized(res)
	}

	pub fn sub(&self, other: &Polynomial) -> Self {
		self.add(&other.negative())
	}

	pub fn mul(&self, other: &Polynomial) -> Self {
		// Get fractions
		let left = self.get_coeffs();
		let right = other.get_coeffs();
		
		// Get res len
		let l_len = left.len();
		let r_len = right.len();
		
		// Init result vector
		let mut res: Vec<Fraction> = vec![Fraction::default();l_len+r_len];

		// Execute double distribution
		for (i, l_frac) in left.iter().enumerate() {
			for (j, r_frac) in right.iter().enumerate() {
				res[i+j] = res[i+j].add(&l_frac.mul(&r_frac));
			}
		}

		Self::normalized(res)
	}

	pub fn div(&self, other: &Polynomial) -> Polynomial {
		// if !other.is_constant() {
		// 	TODO
		// }
		self.mul(&other.reciprocal())
	}

	pub fn pow(&self, power: u32) -> Self {
		let mut res = Self::constant(Fraction::one());
		let mut base = self.clone();
		let mut exp = power;

		while exp > 0 {
			if exp % 2 == 1 {
				res = res.mul(&base);
			}

			exp /= 2;

			if exp > 0 {
				base = base.mul(&base);
			}
		}

		res
	}

	pub fn print(&self) {
		let coeffs = self.get_coeffs();

		for (pow, coeff) in coeffs.iter().enumerate() {
			let mut sign = "";
			if !coeff.is_neg() && pow != 0 {
				sign = "+";
			}

			print!("{}{}X^{} ", sign, coeff.get_number(), pow);
		}
		println!();
	}
}
